Proyectos 1 – 5 unificados
1. Enunciado

Crear un Plan de Mantenimiento que realice diariamente el respaldo completo de la base de datos GradosTítulos.

El script debe incluir:
  • BACKUP DATABASE
  • Compresión
  • Verificación (CHECKSUM)
  • Programación (diaria)
2. Script T‑SQL
-- ========================================================================
--  PROYECTO 1 – Backup completo diario de GradosTítulos
--  Automatización mediante Trabajo (Job) del Agente SQL
-- ========================================================================

USE [msdb];
GO

DECLARE @jobId BINARY(16);

EXEC dbo.sp_add_job
    @job_name           = N'BackupDiario_GradosTitulos',
    @enabled            = 1,
    @description        = N'Backup completo diario de GradosTítulos con compresión y verificación.',
    @start_step_id      = 1,
    @category_name      = N'[Uncategorized (Local)]',
    @owner_login_name   = N'sa';

EXEC dbo.sp_add_jobstep
    @job_id             = @jobId,
    @step_name          = N'Ejecutar BACKUP completo',
    @step_id            = 1,
    @cmdexec_success_code = 0,
    @on_success_action  = 1,
    @on_fail_action     = 2,
    @retry_attempts     = 3,
    @retry_interval     = 5,
    @database_name      = N'master',
    @command            = N'
        BACKUP DATABASE [GradosTítulos]
        TO DISK = N''C:\Backups\GradosTitulos_$(ESCAPE_SQUOTE(SQLINSTANCE))_$(ESCAPE_SQUOTE(DBNAME))_$(ESCAPE_SQUOTE(DATE(yyyyMMdd))).bak''
        WITH COMPRESSION, CHECKSUM, INIT, STATS = 10;
    ';

DECLARE @scheduleId INT;
EXEC dbo.sp_add_schedule
    @schedule_name      = N'Programacion_Diaria_23hs',
    @enabled            = 1,
    @freq_type          = 4,
    @freq_interval      = 1,
    @freq_subday_type   = 1,
    @freq_subday_interval = 0,
    @freq_relative_interval = 0,
    @freq_recurrence_factor = 0,
    @active_start_date  = 20260114,
    @active_end_date    = 99991231,
    @active_start_time  = 230000,
    @active_end_time    = 235959,
    @schedule_id        = @scheduleId OUTPUT;

EXEC dbo.sp_attach_schedule @job_id = @jobId, @schedule_id = @scheduleId;
EXEC dbo.sp_add_jobserver @job_id = @jobId, @server_name = N'(local)';

PRINT N'Trabajo "BackupDiario_GradosTitulos" creado.';
GO
3. Justificación técnica
RequerimientoImplementaciónRazón
BACKUP DATABASEComando nativoEstándar confiable para respaldos.
CompresiónWITH COMPRESSIONReduce tamaño 70‑80%, ahorra espacio y acelera restauración.
VerificaciónWITH CHECKSUMGarantiza integridad del backup.
Programación diariasp_add_schedule con freq_type=4Ejecución automática a las 23:00.
4. Buenas prácticas
  • Uso del Agente SQL en lugar del asistente gráfico (más versionable y depurable).
  • Separación de objetos (job, step, schedule) para fácil mantenimiento.
  • Reintentos (3 intentos, 5 min) para fallos transitorios.
  • Ruta fija y nombres dinámicos con fecha.
  • Script comentado para facilitar la comprensión.
1. Enunciado

Crear un plan semanal para reorganizar índices de las tablas Persona, Tramite, TrabajoInvestigación, Sustentación.

Debe implementar:
  • ALTER INDEX REORGANIZE
  • Job automático
2. Script T‑SQL
-- PROYECTO 2 – Reorganización semanal de índices
USE [msdb]; GO

DECLARE @jobId BINARY(16);
EXEC dbo.sp_add_job
    @job_name = N'ReorganizarIndices_Semanal',
    @enabled = 1,
    @description = N'Reorganiza índices de Persona, Tramite, TrabajoInvestigación y Sustentación cada domingo.',
    @start_step_id = 1,
    @category_name = N'[Uncategorized (Local)]',
    @owner_login_name = N'sa';

EXEC dbo.sp_add_jobstep
    @job_id = @jobId,
    @step_name = N'Reorganizar índices',
    @step_id = 1,
    @cmdexec_success_code = 0,
    @on_success_action = 1,
    @on_fail_action = 2,
    @retry_attempts = 2,
    @retry_interval = 10,
    @database_name = N'GradosTitulos',
    @command = N'
        ALTER INDEX ALL ON [dbo].[Persona] REORGANIZE;
        ALTER INDEX ALL ON [dbo].[Tramite] REORGANIZE;
        ALTER INDEX ALL ON [dbo].[TrabajoInvestigacion] REORGANIZE;
        ALTER INDEX ALL ON [dbo].[Sustentacion] REORGANIZE;
        PRINT ''Índices reorganizados.'';
    ';

DECLARE @scheduleId INT;
EXEC dbo.sp_add_schedule
    @schedule_name = N'Programacion_Semanal_Domingo_02hs',
    @enabled = 1,
    @freq_type = 8,
    @freq_interval = 1,
    @freq_subday_type = 1,
    @freq_recurrence_factor = 1,
    @active_start_date = 20260118,
    @active_start_time = 020000,
    @schedule_id = @scheduleId OUTPUT;

EXEC dbo.sp_attach_schedule @job_id = @jobId, @schedule_id = @scheduleId;
EXEC dbo.sp_add_jobserver @job_id = @jobId, @server_name = N'(local)';
GO

RequerimientoImplementaciónRazón

1. Enunciado

Diseñar un plan inteligente que:

  • Reorganice índices menores al 30%
  • Reconstruya índices mayores al 30%
  • Actualice estadísticas
Utilizar:
  • sys.dm_db_index_physical_stats
  • ALTER INDEX
  • UPDATE STATISTICS
2. Script T‑SQL
-- PROYECTO 3 – Mantenimiento inteligente de índices
USE [GradosTitulos]; GO
SET NOCOUNT ON;

DECLARE @TableName NVARCHAR(255), @IndexName NVARCHAR(255),
        @SchemaName NVARCHAR(255), @Fragmentation FLOAT, @SQL NVARCHAR(MAX);

CREATE TABLE #IndexStats (
    SchemaName NVARCHAR(128), TableName NVARCHAR(128),
    IndexName NVARCHAR(128), Fragmentation FLOAT
);

INSERT INTO #IndexStats
SELECT s.name, o.name, i.name, ips.avg_fragmentation_in_percent
FROM sys.dm_db_index_physical_stats(DB_ID(), NULL, NULL, NULL, 'LIMITED') ips
INNER JOIN sys.objects o ON ips.object_id = o.object_id
INNER JOIN sys.schemas s ON o.schema_id = s.schema_id
INNER JOIN sys.indexes i ON ips.object_id = i.object_id AND ips.index_id = i.index_id
WHERE o.type = 'U' AND o.name IN ('Persona','Tramite','TrabajoInvestigacion','Sustentacion')
  AND i.index_id > 0 AND ips.page_count > 1000;

DECLARE idx_cursor CURSOR FAST_FORWARD FOR
SELECT SchemaName, TableName, IndexName, Fragmentation FROM #IndexStats WHERE Fragmentation > 5;

OPEN idx_cursor;
FETCH NEXT FROM idx_cursor INTO @SchemaName, @TableName, @IndexName, @Fragmentation;

WHILE @@FETCH_STATUS = 0
BEGIN
    IF @Fragmentation >= 5 AND @Fragmentation < 30
        SET @SQL = 'ALTER INDEX [' + @IndexName + '] ON [' + @SchemaName + '].[' + @TableName + '] REORGANIZE;';
    ELSE IF @Fragmentation >= 30
        SET @SQL = 'ALTER INDEX [' + @IndexName + '] ON [' + @SchemaName + '].[' + @TableName + '] REBUILD WITH (ONLINE = ON);';
    EXEC sp_executesql @SQL;
    PRINT 'Acción en ' + @TableName + '.' + @IndexName + ' (Fragmentación: ' + CAST(@Fragmentation AS NVARCHAR) + '%)';
    FETCH NEXT FROM idx_cursor INTO @SchemaName, @TableName, @IndexName, @Fragmentation;
END

CLOSE idx_cursor; DEALLOCATE idx_cursor;

UPDATE STATISTICS [dbo].[Persona] WITH FULLSCAN;
UPDATE STATISTICS [dbo].[Tramite] WITH FULLSCAN;
UPDATE STATISTICS [dbo].[TrabajoInvestigacion] WITH FULLSCAN;
UPDATE STATISTICS [dbo].[Sustentacion] WITH FULLSCAN;

DROP TABLE #IndexStats;
GO
3. Justificación
RequerimientoImplementaciónRazón
DMVsys.dm_db_index_physical_stats con LIMITEDRápido y con bajo overhead.
Reorganizar (<30%)REORGANIZEOperación online, ligera.
Reconstruir (>30%)REBUILD WITH (ONLINE=ON)Elimina fragmentación por completo.
EstadísticasUPDATE STATISTICS ... FULLSCANPrecisión máxima para el optimizador.
4. Buenas prácticas
  • Umbrales estándar (5% y 30%) según Microsoft.
  • Filtrado de índices pequeños (page_count > 1000).
  • Modo LIMITED en DMV para rapidez.
  • Script documentado con prints de seguimiento.
1. Enunciado

Automatizar el mantenimiento completo incluyendo: Backup, CHECKDB, limpieza de historial, reorganización de índices y actualización de estadísticas.

Utilizar Maintenance Plans (simulados) y T‑SQL.
2. Script T‑SQL
-- PROYECTO 4 – Mantenimiento completo (5 pasos)
USE [msdb]; GO

DECLARE @jobId BINARY(16);
EXEC dbo.sp_add_job
    @job_name = N'MantenimientoCompleto_GradosTitulos',
    @enabled = 1,
    @description = N'Mantenimiento completo semanal.',
    @start_step_id = 1,
    @owner_login_name = N'sa';

-- Paso 1: Backup
EXEC dbo.sp_add_jobstep @job_id = @jobId, @step_name = N'01 - Backup', @step_id = 1,
    @command = N'BACKUP DATABASE [GradosTitulos] TO DISK = N''C:\Backups\GradosTitulos_Full_$(ESCAPE_SQUOTE(DATE(yyyyMMdd))).bak'' WITH COMPRESSION, CHECKSUM, INIT;',
    @on_success_action = 3, @on_fail_action = 2;

-- Paso 2: CHECKDB
EXEC dbo.sp_add_jobstep @job_id = @jobId, @step_name = N'02 - CHECKDB', @step_id = 2,
    @database_name = N'GradosTitulos',
    @command = N'DBCC CHECKDB (N''GradosTitulos'') WITH NO_INFOMSGS;',
    @on_success_action = 3, @on_fail_action = 2;

-- Paso 3: Limpieza de historial
EXEC dbo.sp_add_jobstep @job_id = @jobId, @step_name = N'03 - Limpiar historial', @step_id = 3,
    @database_name = N'msdb',
    @command = N'DELETE FROM msdb.dbo.sysjobhistory WHERE run_date < CONVERT(INT, CONVERT(VARCHAR(8), DATEADD(DAY, -30, GETDATE()), 112));',
    @on_success_action = 3, @on_fail_action = 2;

-- Paso 4: Reorganizar índices
EXEC dbo.sp_add_jobstep @job_id = @jobId, @step_name = N'04 - Reorganizar índices', @step_id = 4,
    @database_name = N'GradosTitulos',
    @command = N'
        ALTER INDEX ALL ON [dbo].[Persona] REORGANIZE;
        ALTER INDEX ALL ON [dbo].[Tramite] REORGANIZE;
        ALTER INDEX ALL ON [dbo].[TrabajoInvestigacion] REORGANIZE;
        ALTER INDEX ALL ON [dbo].[Sustentacion] REORGANIZE;',
    @on_success_action = 3, @on_fail_action = 2;

-- Paso 5: Actualizar estadísticas
EXEC dbo.sp_add_jobstep @job_id = @jobId, @step_name = N'05 - Estadísticas', @step_id = 5,
    @database_name = N'GradosTitulos',
    @command = N'EXEC sp_updatestats; UPDATE STATISTICS [dbo].[Persona] WITH FULLSCAN; UPDATE STATISTICS [dbo].[Tramite] WITH FULLSCAN; UPDATE STATISTICS [dbo].[TrabajoInvestigacion] WITH FULLSCAN; UPDATE STATISTICS [dbo].[Sustentacion] WITH FULLSCAN;',
    @on_success_action = 1, @on_fail_action = 2;

-- Programación: domingo 04:00
DECLARE @scheduleId INT;
EXEC dbo.sp_add_schedule @schedule_name = N'Programacion_Semanal_Domingo_04hs',
    @freq_type = 8, @freq_interval = 1, @freq_recurrence_factor = 1,
    @active_start_time = 040000, @schedule_id = @scheduleId OUTPUT;
EXEC dbo.sp_attach_schedule @job_id = @jobId, @schedule_id = @scheduleId;
EXEC dbo.sp_add_jobserver @job_id = @jobId, @server_name = N'(local)';
GO

1. Enunciado

Implementar un plan inteligente que determine automáticamente qué operaciones ejecutar según el nivel de fragmentación, utilizando:

  • Cursores
  • DMV
  • ALTER INDEX
  • UPDATE STATISTICS
  • Tablas de auditoría
2. Script T‑SQL
-- PROYECTO 5 – Plan inteligente con auditoría
USE [GradosTitulos]; GO
SET NOCOUNT ON;

-- 1. Crear tabla de auditoría
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'IndexMaintenanceAudit')
CREATE TABLE dbo.IndexMaintenanceAudit (
    AuditID INT IDENTITY(1,1) PRIMARY KEY,
    ExecutionDate DATETIME DEFAULT GETDATE(),
    SchemaName NVARCHAR(128),
    TableName NVARCHAR(128),
    IndexName NVARCHAR(128),
    FragmentationPct FLOAT,
    ActionTaken NVARCHAR(50),
    CommandExecuted NVARCHAR(1000),
    ErrorMessage NVARCHAR(MAX),
    DurationMs INT
);

-- 2. Cursor con DMV
DECLARE @SchemaName NVARCHAR(128), @TableName NVARCHAR(128),
        @IndexName NVARCHAR(128), @Fragmentation FLOAT,
        @ActionTaken NVARCHAR(50), @SQL NVARCHAR(1000),
        @StartTime DATETIME, @DurationMs INT, @ErrorMessage NVARCHAR(MAX);

DECLARE idx_cursor CURSOR FAST_FORWARD FOR
SELECT s.name, o.name, i.name, ips.avg_fragmentation_in_percent
FROM sys.dm_db_index_physical_stats(DB_ID(), NULL, NULL, NULL, 'LIMITED') ips
INNER JOIN sys.objects o ON ips.object_id = o.object_id
INNER JOIN sys.schemas s ON o.schema_id = s.schema_id
INNER JOIN sys.indexes i ON ips.object_id = i.object_id AND ips.index_id = i.index_id
WHERE o.type = 'U' AND o.name IN ('Persona','Tramite','TrabajoInvestigacion','Sustentacion')
  AND i.index_id > 0 AND ips.page_count > 500 AND ips.avg_fragmentation_in_percent > 5;

OPEN idx_cursor;
FETCH NEXT FROM idx_cursor INTO @SchemaName, @TableName, @IndexName, @Fragmentation;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @StartTime = GETDATE();
    SET @ErrorMessage = NULL;
    SET @ActionTaken = 'NO ACTION';

    BEGIN TRY
        IF @Fragmentation >= 5 AND @Fragmentation < 30
        BEGIN
            SET @ActionTaken = 'REORGANIZE';
            SET @SQL = 'ALTER INDEX [' + @IndexName + '] ON [' + @SchemaName + '].[' + @TableName + '] REORGANIZE;';
        END
        ELSE IF @Fragmentation >= 30
        BEGIN
            SET @ActionTaken = 'REBUILD';
            SET @SQL = 'ALTER INDEX [' + @IndexName + '] ON [' + @SchemaName + '].[' + @TableName + '] REBUILD WITH (ONLINE = ON);';
        END
        ELSE
            SET @ActionTaken = 'NO ACTION';

        IF @ActionTaken <> 'NO ACTION'
            EXEC sp_executesql @SQL;
    END TRY
    BEGIN CATCH
        SET @ErrorMessage = ERROR_MESSAGE();
    END CATCH

    SET @DurationMs = DATEDIFF(MILLISECOND, @StartTime, GETDATE());
    INSERT INTO dbo.IndexMaintenanceAudit (SchemaName, TableName, IndexName, FragmentationPct, ActionTaken, CommandExecuted, ErrorMessage, DurationMs)
    VALUES (@SchemaName, @TableName, @IndexName, @Fragmentation, @ActionTaken, @SQL, @ErrorMessage, @DurationMs);

    FETCH NEXT FROM idx_cursor INTO @SchemaName, @TableName, @IndexName, @Fragmentation;
END

CLOSE idx_cursor; DEALLOCATE idx_cursor;

-- 3. Actualizar estadísticas
UPDATE STATISTICS [dbo].[Persona] WITH FULLSCAN;
UPDATE STATISTICS [dbo].[Tramite] WITH FULLSCAN;
UPDATE STATISTICS [dbo].[TrabajoInvestigacion] WITH FULLSCAN;
UPDATE STATISTICS [dbo].[Sustentacion] WITH FULLSCAN;

-- 4. Reporte de auditoría
SELECT ActionTaken, COUNT(*) AS Cantidad, AVG(FragmentationPct) AS FragmentacionPromedio
FROM dbo.IndexMaintenanceAudit
WHERE ExecutionDate >= CAST(GETDATE() AS DATE)
GROUP BY ActionTaken;
GO

DMVsys.dm_db_index_physical_statsDiagnóstico preciso de fragmentación.
ALTER INDEXREORGANIZE / REBUILDAcción según umbral.

4. Buenas prácticas

  • Filtrado de índices pequeños (page_count > 500).
  • Manejo de errores con TRY...CATCH dentro del cursor.
  • Auditoría detallada (duración, comando, error).
  • Medición de duración por índice.
