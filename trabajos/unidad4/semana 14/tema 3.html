1. Enunciado

Crear un procedimiento almacenado que genere automáticamente un respaldo completo y posteriormente sea ejecutado mediante PowerShell.

El script debe desarrollar:
  • Stored Procedure
  • BACKUP DATABASE
  • PowerShell Invoke-Sqlcmd
2. Script T‑SQL · Stored Procedure
-- ========================================================================
--  TEMA 3 – PROYECTO 1
--  Procedimiento almacenado para backup completo de GradosTitulos
--  Incluye compresión, verificación (CHECKSUM) y nombre dinámico
-- ========================================================================

USE [GradosTitulos];
GO

IF OBJECT_ID('dbo.usp_BackupCompleto', 'P') IS NOT NULL
    DROP PROCEDURE dbo.usp_BackupCompleto;
GO

CREATE PROCEDURE dbo.usp_BackupCompleto
    @BackupPath NVARCHAR(500) = N'C:\Backups\',
    @DatabaseName NVARCHAR(128) = N'GradosTitulos',
    @WithCompression BIT = 1,
    @WithChecksum BIT = 1,
    @BackupFileName NVARCHAR(255) = NULL OUTPUT,
    @ErrorMessage NVARCHAR(MAX) = NULL OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        DECLARE @DirCommand NVARCHAR(500) = 'EXEC xp_create_subdir ''' + @BackupPath + ''';';
        EXEC sp_executesql @DirCommand;

        DECLARE @Timestamp NVARCHAR(20) = FORMAT(GETDATE(), 'yyyyMMdd_HHmmss');
        SET @BackupFileName = @BackupPath + @DatabaseName + '_Full_' + @Timestamp + '.bak';

        DECLARE @SQL NVARCHAR(MAX) = 'BACKUP DATABASE [' + @DatabaseName + '] TO DISK = N''' + @BackupFileName + ''' WITH INIT, STATS = 10';

        IF @WithCompression = 1 SET @SQL = @SQL + ', COMPRESSION';
        IF @WithChecksum = 1 SET @SQL = @SQL + ', CHECKSUM';

        EXEC sp_executesql @SQL;
    END TRY
    BEGIN CATCH
        SET @ErrorMessage = ERROR_MESSAGE();
        RAISERROR(@ErrorMessage, 16, 1);
    END CATCH
END;
GO
SP parametrizado con creación automática de carpeta y nombre con timestamp.
3. Script PowerShell · Invoke-Sqlcmd
# TEMA 3 – PROYECTO 1
# Script PowerShell para ejecutar el SP de backup

$ServerInstance = "localhost"
$Database      = "GradosTitulos"
$BackupPath    = "C:\Backups\"

$SqlCommand = @"
    EXEC dbo.usp_BackupCompleto
        @BackupPath = N'$BackupPath',
        @DatabaseName = N'GradosTitulos',
        @WithCompression = 1,
        @WithChecksum = 1;
"@

try {
    Invoke-Sqlcmd -ServerInstance $ServerInstance -Database $Database -Query $SqlCommand -QueryTimeout 0
    Write-Host "✅ Backup completado." -ForegroundColor Green
} catch {
    Write-Host "❌ Error: $($_.Exception.Message)" -ForegroundColor Red
}

1. Enunciado

Automatizar la exportación diaria de los trámites aprobados hacia un archivo CSV utilizando PowerShell.

Utilizar:
  • SELECT
  • Invoke-Sqlcmd
  • Export-Csv
2. Script PowerShell
# TEMA 3 – PROYECTO 2
$ServerInstance = "localhost"
$Database      = "GradosTitulos"
$OutputPath    = "C:\Exports\"
$DateStamp     = Get-Date -Format "yyyyMMdd"
$FullPath      = Join-Path $OutputPath "TramitesAprobados_$DateStamp.csv"

$SqlQuery = @"
    SELECT Id, NumeroTramite, FechaSolicitud, FechaAprobacion, PersonaId, TipoTramite, Estado
    FROM dbo.Tramite
    WHERE Estado = 'Aprobado' AND CAST(FechaAprobacion AS DATE) = CAST(GETDATE() AS DATE)
"@

try {
    $Data = Invoke-Sqlcmd -ServerInstance $ServerInstance -Database $Database -Query $SqlQuery
    if ($Data) {
        $Data | Export-Csv -Path $FullPath -NoTypeInformation -Encoding UTF8
        Write-Host "✅ Exportados $($Data.Count) registros." -ForegroundColor Green
    } else {
        Write-Host "⚠️ No hay trámites aprobados hoy." -ForegroundColor Yellow
    }
} catch {
    Write-Host "❌ Error: $($_.Exception.Message)" -ForegroundColor Red
}

1. Enunciado

Desarrollar un proceso automático que consulte diariamente la cantidad de trámites, genere un archivo Excel y almacene una copia en una carpeta de respaldo.

Utilizar:
  • T‑SQL
  • PowerShell
  • Export-Excel (ImportExcel)
2. T‑SQL (consulta)
-- TEMA 3 – PROYECTO 3 – Consulta de resumen diario
SELECT Estado, COUNT(*) AS Cantidad,
    MIN(FechaSolicitud) AS PrimerTramite,
    MAX(FechaSolicitud) AS UltimoTramite,
    CAST(GETDATE() AS DATE) AS FechaReporte
FROM dbo.Tramite
WHERE CAST(FechaSolicitud AS DATE) = CAST(GETDATE() AS DATE)
GROUP BY Estado;
3. PowerShell (Export-Excel)
# TEMA 3 – PROYECTO 3
if (-not (Get-Module -Name ImportExcel -ListAvailable)) {
    Install-Module -Name ImportExcel -Scope CurrentUser -Force
}
Import-Module -Name ImportExcel -Force

$ServerInstance = "localhost"
$Database      = "GradosTitulos"
$OutputFolder  = "C:\Reportes\"
$BackupFolder  = "C:\BackupReportes\"
$DateStamp     = Get-Date -Format "yyyyMMdd"
$FileName      = "ReporteTramites_$DateStamp.xlsx"
$FullPath      = Join-Path $OutputFolder $FileName
$BackupPath    = Join-Path $BackupFolder $FileName

$SqlQuery = @"
    SELECT Estado, COUNT(*) AS Cantidad,
           MIN(FechaSolicitud) AS PrimerTramite,
           MAX(FechaSolicitud) AS UltimoTramite,
           CAST(GETDATE() AS DATE) AS FechaReporte
    FROM dbo.Tramite
    WHERE CAST(FechaSolicitud AS DATE) = CAST(GETDATE() AS DATE)

"@

    $Data = Invoke-Sqlcmd -ServerInstance $ServerInstance -Database $Database -Query $SqlQuery

1. Enunciado

Implementar una solución de automatización completa que ejecute respaldos, valide con CHECKDB, genere reportes, comprima los respaldos y elimine los mayores a 30 días.

Integrar:
  • SQL Server
  • PowerShell
  • compresión ZIP
  • registro de auditoría
  • manejo de errores
2. Script PowerShell (completo)
# TEMA 3 – PROYECTO 5 – Automatización integral
$ServerInstance = "localhost"
$Database      = "GradosTitulos"
$BackupRoot    = "C:\Backups\"
$LogPath       = "C:\Logs\BackupAudit.log"
$RetentionDays = 30
$DateStamp     = Get-Date -Format "yyyyMMdd_HHmmss"
$BackupFolder  = Join-Path $BackupRoot $DateStamp
$BackupFile    = Join-Path $BackupFolder "$Database.bak"
$ZipFile       = Join-Path $BackupFolder "$Database.zip"

function Write-Log {
    param ([string]$Message, [string]$Level = "INFO")
    $Timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    Add-Content -Path $LogPath -Value "[$Timestamp] [$Level] $Message"
    Write-Host "[$Timestamp] [$Level] $Message" -ForegroundColor if ($Level -eq "ERROR") { Red } else { Cyan }
}

try {
    Write-Log "========== INICIO =========="
    New-Item -ItemType Directory -Path $BackupFolder -Force | Out-Null

    # Backup
    Write-Log "Ejecutando BACKUP..."
    $BackupQuery = "BACKUP DATABASE [$Database] TO DISK = N'$BackupFile' WITH COMPRESSION, CHECKSUM, INIT, STATS = 10;"
    Invoke-Sqlcmd -ServerInstance $ServerInstance -Query $BackupQuery -QueryTimeout 0

    # CHECKDB
    Write-Log "Ejecutando CHECKDB..."
    Invoke-Sqlcmd -ServerInstance $ServerInstance -Database $Database -Query "DBCC CHECKDB (N'$Database') WITH NO_INFOMSGS;" -QueryTimeout 0

    # Compresión ZIP
    Write-Log "Comprimiendo a ZIP..."
    Compress-Archive -Path $BackupFile -DestinationPath $ZipFile -CompressionLevel Optimal

    # Eliminar antiguos
    Write-Log "Eliminando respaldos > $RetentionDays días..."
    $CutoffDate = (Get-Date).AddDays(-$RetentionDays)
    Get-ChildItem -Path $BackupRoot -Directory | Where-Object { $_.CreationTime -lt $CutoffDate } | Remove-Item -Recurse -Force

    # Reporte
    $ReportFile = Join-Path $BackupFolder "Reporte.txt"
    $ReportContent = @"
Reporte de Respaldo
Fecha: $(Get-Date)
Servidor: $ServerInstance
Base de datos: $Database
Backup: $BackupFile
ZIP: $ZipFile
CHECKDB: Exitoso
Retención: $RetentionDays días
"@
    Set-Content -Path $ReportFile -Value $ReportContent

    Write-Log "========== PROCESO COMPLETADO =========="
} catch {
    Write-Log "ERROR: $($_.Exception.Message)" -Level "ERROR"
    Write-Log "========== PROCESO FALLIDO ==========" -Level "ERROR"
}
